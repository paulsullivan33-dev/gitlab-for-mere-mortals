$ chmod +x gitlab-runner

# Runners are created, not registered with a shared token
# (the old registration-token flow is gone in current
# GitLab). Creating one through the API returns its own
# authentication token:

$ curl -s -X POST -H "PRIVATE-TOKEN: $TOK" \
    --data-urlencode "runner_type=instance_type" \
    --data-urlencode "description=lab shell runner" \
    --data-urlencode "tag_list=lab-shell" \
    http://127.0.0.1:8929/api/v4/user/runners
runner id: 1

$ ./gitlab-runner register --non-interactive \
    --config /home/hatch/workspace/gitlab-labwork/runner-config.toml \
    --url http://127.0.0.1:8929 --token "$RTOK" \
    --executor shell --description "lab shell runner"
Verifying runner... is valid
Runner registered successfully. Feel free to start it, but if it's running already the config should be automatically reloaded!
Configuration (with the authentication token) was saved in "/home/hatch/workspace/gitlab-labwork/runner-config.toml"

$ ./gitlab-runner run --config /home/hatch/workspace/gitlab-labwork/runner-config.toml
